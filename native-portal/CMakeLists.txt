cmake_minimum_required(VERSION 3.16)
project(ft_native_portal_math LANGUAGES CXX)

add_library(ft-aperture STATIC src/aperture.cpp)
target_include_directories(ft-aperture PUBLIC include)
target_compile_features(ft-aperture PUBLIC cxx_std_17)
target_compile_options(ft-aperture PRIVATE -Wall -Wextra -Werror)

include(CTest)
if(BUILD_TESTING)
  add_executable(test-aperture tests/test_aperture.cpp)
  target_link_libraries(test-aperture PRIVATE ft-aperture)
  target_compile_options(test-aperture PRIVATE -Wall -Wextra -Werror)
  add_test(NAME aperture COMMAND test-aperture)
endif()

option(FT_PORTAL_GPU_TESTS "Build owned offscreen Vulkan test, never native runtime hooks" OFF)
set(FT_PORTAL_GPU_SHADER_DIR "${CMAKE_CURRENT_SOURCE_DIR}/build/gpu" CACHE PATH "Owned shaders generated by tests/gpu_shaders.py")
if(FT_PORTAL_GPU_TESTS)
  find_package(Vulkan REQUIRED)
  add_executable(test-gpu tests/test_gpu.cpp)
  target_link_libraries(test-gpu PRIVATE ft-aperture Vulkan::Vulkan)
  target_compile_options(test-gpu PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
  if(BUILD_TESTING)
    add_test(NAME aperture-gpu COMMAND test-gpu "${FT_PORTAL_GPU_SHADER_DIR}")
  endif()
endif()
